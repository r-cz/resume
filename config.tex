\hypersetup{hidelinks}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%     Layout overrides
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%% Summary paragraph in the body font (this class version has no cvparagraph)
\newenvironment{cvsummary}{%
  \par\noindent\begin{minipage}{\textwidth}
  \fontsize{10pt}{1.2em}\bodyfontlight\upshape\color{text}\selectfont
}{%
  \end{minipage}\par\vspace{1mm}
}

%%% Skills table sized for the page width (the class hard-codes a 15.2cm column,
%%% which overflows the page once a skill label is longer than a couple of words)
\renewenvironment{cvskills}{%
  \begin{center}
    \setlength\tabcolsep{1ex}
    \setlength{\extrarowheight}{0pt}
    \begin{tabular*}{\textwidth}{@{\extracolsep{\fill}} l L{\dimexpr\textwidth-3.6cm\relax}}
}{%
    \end{tabular*}
  \end{center}
}

%%% A little breathing room between experience entries
\renewenvironment{cvitems}{%
  \vspace{-4mm}
  \begin{justify}
  \begin{itemize}[leftmargin=2ex, nosep, noitemsep]
    \setlength{\parskip}{0pt}
    \renewcommand{\labelitemi}{\bullet}
}{%
  \end{itemize}
  \end{justify}
  \vspace{0.5mm}
}
